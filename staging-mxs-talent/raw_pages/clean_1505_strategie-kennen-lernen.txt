document.addEventListener('DOMContentLoaded', function() {
    /**
     * Scrollt sanft zum angegebenen Element.
     * @param {HTMLElement} element - Das Zielelement, zu dem gescrollt werden soll.
     */
    function smoothScrollTo(element) {
        element.scrollIntoView({ behavior: 'smooth' });
    }

    /**
     * Ruft den Wert eines URL-Parameters ab.
     * @param {string} name - Der Name des abzurufenden URL-Parameters.
     * @returns {string} Der Wert des URL-Parameters oder ein leerer String, wenn nicht gefunden.
     */
    function getUrlParameter(name) {
        name = name.replace(/[\[]/, '\\[').replace(/[\]]/, '\\]');
        var regex = new RegExp('[\\?&]' + name + '=([^&#]*)');
        var results = regex.exec(location.search);
        return results === null ? '' : decodeURIComponent(results[1].replace(/\+/g, ' '));
    }

    // Holt die Ziel-Element-ID aus dem URL-Parameter
    var scrollTarget = getUrlParameter('scrollTo');

    if (scrollTarget) {
        var targetElement = document.getElementById(scrollTarget);
        if (targetElement) {
            // Wartet, bis alle Bilder und Ressourcen geladen sind
            window.addEventListener('load', function() {
                // Verzögert das Scrollen um eine kurze Zeit
                setTimeout(function() {
                    smoothScrollTo(targetElement);

                    // Stellt sicher, dass die Position beibehalten wird
                    var checkPosition = setInterval(function() {
                        var elementPosition = targetElement.getBoundingClientRect().top;
                        if (Math.abs(elementPosition) > 5) {
                            window.scrollTo(0, window.pageYOffset + elementPosition);
                        } else {
                            clearInterval(checkPosition);
                        }
                    }, 100);
                }, 500);
            });
        }
    }
});
